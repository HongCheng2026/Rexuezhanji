## 3. 可直接粘贴进程序的代码

```js
// commanderExpConfig.js

export const COMMANDER_MAX_LEVEL = 60;

export const COMMANDER_EXP_TO_NEXT_LEVEL = [
  0, 130, 190, 224, 246, 266, 282, 298, 310, 322,
  334, 344, 354, 362, 370, 378, 386, 394, 400, 410,
  1000, 1100, 1200, 1300, 1400, 1000, 1100, 1200, 1300, 1400,
  2736, 3548, 3938, 4214, 4432, 4614, 4772, 4910, 5034, 5148,
  5252, 5348, 5438, 5524, 5602, 5678, 5750, 5818, 5882, 5944,
  6004, 6062, 6118, 6172, 6222, 6272, 6322, 6370, 6416, 6460,
  0
];

export const COMMANDER_TOTAL_EXP_BY_LEVEL = [
  0, 0, 130, 320, 544, 790, 1056, 1338, 1636, 1946,
  2268, 2602, 2946, 3300, 3662, 4032, 4410, 4796, 5190, 5590,
  6000, 7000, 8100, 9300, 10600, 12000, 13000, 14100, 15300, 16600,
  18000, 20736, 24284, 28222, 32436, 36868, 41482, 46254, 51164, 56198,
  61346, 66598, 71946, 77384, 82908, 88510, 94188, 99938, 105756, 111638,
  117582, 123586, 129648, 135766, 141938, 148160, 154432, 160754, 167124, 173540,
  180000
];

export function getCommanderLevel(totalExp) {
  let level = 1;

  for (let currentLevel = 1; currentLevel < COMMANDER_MAX_LEVEL; currentLevel++) {
    const needExp = COMMANDER_EXP_TO_NEXT_LEVEL[currentLevel];

    if (totalExp < needExp) {
      break;
    }

    totalExp -= needExp;
    level++;
  }

  return level;
}

export function getCommanderExpProgress(totalExp) {
  let level = 1;
  let expLeft = totalExp;

  for (let currentLevel = 1; currentLevel < COMMANDER_MAX_LEVEL; currentLevel++) {
    const needExp = COMMANDER_EXP_TO_NEXT_LEVEL[currentLevel];

    if (expLeft < needExp) {
      return {
        level,
        currentExp: expLeft,
        needExp,
        progress: needExp <= 0 ? 1 : expLeft / needExp,
        isMaxLevel: false,
        overflowExp: 0
      };
    }

    expLeft -= needExp;
    level++;
  }

  return {
    level: COMMANDER_MAX_LEVEL,
    currentExp: expLeft,
    needExp: 100000,
    progress: (expLeft % 100000) / 100000,
    isMaxLevel: true,
    overflowExp: expLeft
  };
}

export function getTotalExpRequiredForLevel(level) {
  const safeLevel = Math.max(1, Math.min(level, COMMANDER_MAX_LEVEL));
  return COMMANDER_TOTAL_EXP_BY_LEVEL[safeLevel];
}
```
