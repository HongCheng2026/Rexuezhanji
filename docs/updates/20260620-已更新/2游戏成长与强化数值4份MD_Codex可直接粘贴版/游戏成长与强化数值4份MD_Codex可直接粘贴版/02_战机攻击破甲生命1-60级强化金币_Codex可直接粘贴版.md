# 02_战机攻击破甲生命1-60级强化金币_Codex可直接粘贴版

## 1. 固定口径

```text
战机强化等级上限 = 指挥官等级
战机最高强化等级 = 60级

战机强化分为三条独立强化线：
1. 攻击
2. 破甲
3. 生命

金币成本分配：
攻击 = 本级总强化成本的30%
破甲 = 本级总强化成本的45%
生命 = 本级总强化成本的25%

强化到20级累计总成本 = 60000金币
强化到25级累计总成本 = 120000金币
强化到30级累计总成本 = 180000金币
强化到60级累计总成本 = 1800000金币
```

## 2. 每级属性成长

```text
攻击每强化1级：攻击力 +1
破甲每强化1级：破甲率 +0.1%
生命每强化1级：生命值 +10

0.1% 在程序里写作 0.001
```

export function getFighterUpgradeCost(statType, targetLevel) {
  if (targetLevel < 1 || targetLevel > FIGHTER_MAX_UPGRADE_LEVEL) {
    return null;
  }

  const costTableMap = {
    attack: FIGHTER_ATTACK_UPGRADE_COST_BY_TARGET_LEVEL,
    armorPenetration: FIGHTER_ARMOR_PENETRATION_UPGRADE_COST_BY_TARGET_LEVEL,
    hp: FIGHTER_HP_UPGRADE_COST_BY_TARGET_LEVEL
  };

  const table = costTableMap[statType];

  if (!table) {
    return null;
  }

  return table[targetLevel];
}

export function getFighterUpgradeTotalCost(targetLevel) {
  if (targetLevel < 1 || targetLevel > FIGHTER_MAX_UPGRADE_LEVEL) {
    return null;
  }

  return FIGHTER_UPGRADE_TOTAL_COST_BY_TARGET_LEVEL[targetLevel];
}

export function getFighterUpgradeCumulativeTotalCost(level) {
  const safeLevel = Math.max(1, Math.min(level, FIGHTER_MAX_UPGRADE_LEVEL));
  return FIGHTER_UPGRADE_CUMULATIVE_TOTAL_COST_BY_LEVEL[safeLevel];
}

export function getFighterUpgradeStats({
  attackLevel,
  armorPenetrationLevel,
  hpLevel
}) {
  return {
    attackBonus: attackLevel * FIGHTER_UPGRADE_STAT_GAIN.attackPerLevel,
    armorPenetrationBonus:
      armorPenetrationLevel * FIGHTER_UPGRADE_STAT_GAIN.armorPenetrationPerLevel,
    hpBonus: hpLevel * FIGHTER_UPGRADE_STAT_GAIN.hpPerLevel
  };
}

export function canUpgradeFighterStat({
  commanderLevel,
  currentStatLevel,
  gold,
  statType
}) {
  const targetLevel = currentStatLevel + 1;

  if (targetLevel > commanderLevel) {
    return {
      canUpgrade: false,
      reason: "COMMANDER_LEVEL_NOT_ENOUGH",
      targetLevel
    };
  }

  if (targetLevel > FIGHTER_MAX_UPGRADE_LEVEL) {
    return {
      canUpgrade: false,
      reason: "MAX_LEVEL",
      targetLevel
    };
  }

  const cost = getFighterUpgradeCost(statType, targetLevel);

  if (cost === null) {
    return {
      canUpgrade: false,
      reason: "UNKNOWN_STAT_TYPE",
      targetLevel
    };
  }

  if (gold < cost) {
    return {
      canUpgrade: false,
      reason: "GOLD_NOT_ENOUGH",
      targetLevel,
      cost
    };
  }

  return {
    canUpgrade: true,
    targetLevel,
    cost
  };
}

export function upgradeFighterStat(player, statType) {
  const levelFieldMap = {
    attack: "fighterAttackLevel",
    armorPenetration: "fighterArmorPenetrationLevel",
    hp: "fighterHpLevel"
  };

  const levelField = levelFieldMap[statType];

  if (!levelField) {
    return {
      success: false,
      reason: "UNKNOWN_STAT_TYPE"
    };
  }

  const check = canUpgradeFighterStat({
    commanderLevel: player.commanderLevel,
    currentStatLevel: player[levelField],
    gold: player.gold,
    statType
  });

  if (!check.canUpgrade) {
    return {
      success: false,
      reason: check.reason,
      targetLevel: check.targetLevel,
      cost: check.cost || 0
    };
  }

  player.gold -= check.cost;
  player[levelField] = check.targetLevel;

  return {
    success: true,
    statType,
    targetLevel: check.targetLevel,
    cost: check.cost,
    gold: player.gold
  };
}
