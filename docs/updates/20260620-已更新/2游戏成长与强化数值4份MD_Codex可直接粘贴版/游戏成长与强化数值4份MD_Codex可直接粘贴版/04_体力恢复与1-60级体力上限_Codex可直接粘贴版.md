# 04_体力恢复与1-60级体力上限_Codex可直接粘贴版

## 1. 固定口径

```text
初始体力上限：300
1级体力上限：305
60级体力上限：600

体力上限公式：
体力上限 = 300 + 当前等级 * 5

每5分钟恢复1点体力。
也就是每分钟恢复0.2点体力。
```

## 3. 可直接粘贴进程序的代码

```js
// staminaConfig.js

export const STAMINA_CONFIG = {
  baseMaxStamina: 300,
  maxLevel: 60,
  staminaCapPerLevel: 5,

  recoverAmount: 1,
  recoverIntervalSeconds: 5 * 60,
  recoverPerMinute: 0.2
};

export const STAMINA_CAP_BY_LEVEL = [
  0, 305, 310, 315, 320, 325, 330, 335, 340, 345,
  350, 355, 360, 365, 370, 375, 380, 385, 390, 395,
  400, 405, 410, 415, 420, 425, 430, 435, 440, 445,
  450, 455, 460, 465, 470, 475, 480, 485, 490, 495,
  500, 505, 510, 515, 520, 525, 530, 535, 540, 545,
  550, 555, 560, 565, 570, 575, 580, 585, 590, 595,
  600
];

export function getMaxStaminaByLevel(level) {
  const safeLevel = Math.max(1, Math.min(level, STAMINA_CONFIG.maxLevel));
  return STAMINA_CAP_BY_LEVEL[safeLevel];
}

export function getRecoverableStaminaByOfflineSeconds(offlineSeconds) {
  return Math.floor(
    offlineSeconds / STAMINA_CONFIG.recoverIntervalSeconds
  ) * STAMINA_CONFIG.recoverAmount;
}

export function recoverStamina({
  currentStamina,
  level,
  offlineSeconds
}) {
  const maxStamina = getMaxStaminaByLevel(level);
  const recoverAmount = getRecoverableStaminaByOfflineSeconds(offlineSeconds);

  return Math.min(maxStamina, currentStamina + recoverAmount);
}

export function getStaminaRecoverPerMinute() {
  return STAMINA_CONFIG.recoverPerMinute;
}

export function getStaminaCapIncreaseOnLevelUp({
  oldLevel,
  newLevel
}) {
  const oldCap = getMaxStaminaByLevel(oldLevel);
  const newCap = getMaxStaminaByLevel(newLevel);

  return {
    oldLevel,
    newLevel,
    oldCap,
    newCap,
    increase: newCap - oldCap
  };
}
```
